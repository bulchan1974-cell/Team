SQL Script สำหรับสร้างตารางใน Supabase SQL Editor

คัดลอกไปรันในเมนู SQL Editor ของโปรเจกต์ Supabase 

-- สร้างตาราง students สำหรับระบบลงทะเบียนนักศึกษา วิทยาลัยเทคนิคนางรอง
create table if not exists public.students (
  id uuid primary key default gen_random_uuid(),
  created_at timestamptz not null default now(),
  student_code text not null unique check (char_length(student_code) = 11),
  prefix text not null check (prefix in ('นาย', 'นางสาว', 'นาง')),
  full_name text not null,
  national_id text not null unique check (char_length(national_id) = 13),
  birth_date date not null,
  phone text not null,
  email text not null,
  education_level text not null check (education_level in ('ปวช.1', 'ปวช.2', 'ปวช.3', 'ปวส.1', 'ปวส.2')),
  department text not null,
  address text not null,
  guardian_name text not null,
  guardian_phone text not null
);

-- สร้าง Index เพื่อช่วยให้ค้นหาและเรียงลำดับได้รวดเร็ว
create index if not exists idx_students_code on public.students (student_code);
create index if not exists idx_students_department on public.students (department);

-- เปิดใช้งาน Row Level Security (RLS)
alter table public.students enable row level security;

-- Policy เบื้องต้นสำหรับช่วงพัฒนา (อนุญาตให้ anon อ่าน/เพิ่ม/ลบ ข้อมูลได้)
-- คำแนะนำ: เมื่อใช้งานจริงในระบบโปรดักชัน ควรปรับ Policy ให้รัดกุมขึ้นตามบทบาทผู้ใช้
drop policy if exists "Allow anon select students" on public.students;
create policy "Allow anon select students"
  on public.students for select
  to anon
  using (true);

drop policy if exists "Allow anon insert students" on public.students;
create policy "Allow anon insert students"
  on public.students for insert
  to anon
  with check (true);

drop policy if exists "Allow anon delete students" on public.students;
create policy "Allow anon delete students"
  on public.students for delete
  to anon
  using (true);

ขั้นตอนการเชื่อมต่อ Supabase:

  1. สร้างโปรเจกต์ใหม่ที่ supabase.com แล้วเปิดเมนู SQL Editor
  2. วางโค้ด SQL ด้านบนแล้วกด Run เพื่อสร้างตาราง public.students และ Policy
  3. ไปที่ Project Settings → API คัดลอก Project URL และ anon public key มาใส่ช่องด้านซ้าย
  4. กด บันทึกการตั้งค่า และ ทดสอบการเชื่อมต่อ ระบบจะสลับไปใช้งานฐานข้อมูลจริงทันที
